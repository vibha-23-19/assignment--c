2.Write a function swapPlaylistCounts(int *a, int *b) that swaps the number of songs in two Spotify playlists using 
  pointers, then call the function in main and print the swapped values.
  
  #include <stdio.h>

void swapPlaylistCounts(int *a, int *b)
{
    int temp;
    
    temp = *a;
    *a = *b;
    *b = temp;
    
}
int main()
{
    int playlist1 = 25;
    int playlist2 = 50;
    
    printf("\n before swapping");
    printf("\n playlist 1 = %d songs" , playlist1);
    printf("\n playlist 2 = %d songs" , playlist2);
    
    swapPlaylistCounts(&playlist1, &playlist2);
    
    printf("\n after swapping");
    printf("\n playlist 1 = %d songs" , playlist1);
    printf("\n playlist 2 = %d songs" , playlist2);
	
	return 0;
    
}

3.Given an array of 5 order amounts (e.g., Zomato orders), use a pointer to iterate through the array and print
  each amount along with its memory address.<br><br><em><strong>Hint:</strong> Use pointer arithmetic to move to the next element.</em>
  #include <stdio.h>

int main()
{
    int orders[5] = {200, 170, 350, 250, 500};

    int *pointer = orders;

    for(int i = 0; i < 5; i++)
    {
        printf("\nOrder Amount: %d", *pointer);
        printf("\nMemory Address: %p\n", &orders);

        pointer++;
    }

    return 0;
}

4.Create a function incrementFollowers(int *followers, int n) that increases each follower count in an array 
 (representing Instagram followers for 5 friends) by 100 using pointer arithmetic, then print the updated counts.
 
#include <stdio.h>

void incrementFollowers(int *followers, int n)
{
    for(int i = 0; i < n; i++)
    {
        *followers = *followers + 100;
        followers++;
    }
}

int main()
{
    int followers[5] = {400, 700, 1000, 1200, 1500};

    incrementFollowers(followers, 5);

    printf("\nUpdated follower counts:");

    for(int i = 0; i < 5; i++)
    {
        printf("%d\n", followers[i]);
    }

    return 0;
}
