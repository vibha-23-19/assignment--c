2.Create a constant variable to store the GST rate (for example, 18%) and use it to calculate 
  the final price of a Zomato order with a given base price.
  
  #include <stdio.h>

   int main() {
    const float gstrate = 0.18;  
    float basePrice = 500.0;
    float finalPrice = basePrice + (basePrice * gstrate);

    printf("Base Price: %.2f\n", basePrice);
    printf("GST Amount: %.2f\n", basePrice * gstrate);
    printf("Final Price with GST -> %.2f\n", finalPrice);

    return 0;
}
