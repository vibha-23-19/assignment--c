2.Build a Zomato-style food suggestion tool: take the user's preferred meal time ('breakfast', 'lunch', 'dinner', or 'snack') 
  and use a switch-case statement to suggest a popular dish for that time. If the input doesn't match any meal, suggest 'Try some fruits!'.
  
 #include <stdio.h>

int main() {
    
    int choice;

    printf("Enter your meal time \n 1) breakfast \n 2) lunch \n 3) dinner \n 4) snack \n");
    scanf("%d", &choice);

    switch (choice) 
    {
        case 1:
            printf("Try Fafda & Jalebi!\n");
            break;
        case 2:
            printf("Try Gujrati Thali!\n");
            break;
        case 3:
            printf("Try Paneer Tikka!\n");
            break;
        case 4:
            printf("Try Samosa or Pakora!\n");
            break;
        default:
            printf("Try some fruits!\n");
    }

    return 0;
}

3.Create a Flipkart discount calculator that asks the user for the total cart amount. Use nested if statements to check: 
if amount > 2000, apply 20% discount; else if amount > 1000, apply 10% discount; else, no discount. Print the final amount to pay.
<br><br><em><strong>Hint:</strong> Use nested ifs to check each discount slab.</em>
 
 #include <stdio.h>

 int main() {
    float amount, discount, finalamount;

    printf("Enter total cart amount: ");
    scanf("%f", &amount);

    if (amount > 2000) {
        discount = amount * 20/100;
        finalamount = amount - discount;
        printf("20%% discount applied.\n");
    } else {
        if (amount > 1000) {
            discount = amount * 10/100;
            finalamount = amount - discount;
            printf("10%% discount applied.\n");
        } else {
            discount = 0;
            finalamount = amount;
            printf("No discount applied.\n");
        }
    }

    printf("Discount amount: %.2f\n", discount);
    printf("Final amount to pay: %.2f\n", finalamount);

    return 0;
}
