3.Build a function that takes a 1D array of 7 integers representing your daily Zomato order amounts and 
calculates the average spend for the week.<br><br><em><strong>Hint:</strong> Use a loop to sum the values, 
then divide by the array length.</em>

#include<stdio.h>

float Calculator(int a[])
{
    int i, sum=0;
  
   for(i=0;i<7;i++)
  {
   sum=sum+a[i];
  }
 
  return (float)sum/7;

 }

int main()
{
    int amount[7];
   
    for(int i=0;i<7;i++)
    {
        printf("Enter amount for day %d: ", i + 1);
        scanf("%d", &amount[i]);
    }

    printf("Average = %f", Calculator(amount));

    return 0;
}
